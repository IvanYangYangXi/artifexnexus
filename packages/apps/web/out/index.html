import bpy

def create_sphere(radius=1.0, location=(0, 0, 0)):
    """在 Blender 中创建一个球体"""
    bpy.ops.mesh.primitive_uv_sphere_add(
        radius=radius,
        location=location
    )
    return bpy.context.active_object

# 创建球体
sphere = create_sphere(radius=2.0)
print(f"创建球体: {sphere.name}")
